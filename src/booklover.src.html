-- D30 retention by monthly cohort and channel
WITH cohort AS (
  SELECT id AS user_id,
         DATE_TRUNC('month', created_at) AS cohort_month,
         channel,
         created_at
  FROM users
),
active_d30 AS (
  SELECT DISTINCT c.user_id
  FROM cohort c
  JOIN events e ON e.user_id = c.user_id
  WHERE e.event_time >= c.created_at + INTERVAL '28 days'
    AND e.event_time <  c.created_at + INTERVAL '31 days'
)
SELECT c.cohort_month,
       c.channel,
       COUNT(*)                                       AS new_users,
       COUNT(a.user_id)                               AS active_d30,
       ROUND(100.0 * COUNT(a.user_id) / COUNT(*), 1)  AS retention_d30
FROM cohort c
LEFT JOIN active_d30 a ON a.user_id = c.user_id
GROUP BY 1, 2
ORDER BY 1, retention_d30;
